name: Build Android APK
on:
  workflow_dispatch:
  push:
    paths:
      - 'android/**'
      - 'index.html'
      - 'style.css'
      - 'app.js'
      - 'manifest.json'
      - 'assets/**'
      - '.github/workflows/build-apk.yml'
jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'
      - uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '8.7'
      - uses: android-actions/setup-android@v3
      - name: Install Android SDK platform and build-tools
        run: sdkmanager "platforms;android-35" "build-tools;35.0.0"
      - name: Sync web app into Android WebView assets
        run: |
          mkdir -p android/app/src/main/assets/www
          cp index.html style.css app.js manifest.json sw.js android/app/src/main/assets/www/
          mkdir -p android/app/src/main/assets/www/assets
          cp assets/* android/app/src/main/assets/www/assets/
      - name: Build debug APK
        working-directory: android
        run: gradle assembleDebug
      - uses: actions/upload-artifact@v4
        with:
          name: SISTEMA-Despertar-Android-APK
          path: android/app/build/outputs/apk/debug/app-debug.apk
          retention-days: 30
